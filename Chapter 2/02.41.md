- _[...] write a program that reads a set of book sales transactions, writing each transaction to the standard output_. **[(1.20)](https://github.com/Stachu72/CPP_Primer_Solutions/blob/main/Chapter%201/01.20.md)**:
```c++
#include <iostream>
#include <string>

struct Sales_data {
    std::string ISBN;
    unsigned copies_sold = 0;
    double price = 0.0;
};

int main()
{
    Sales_data SalesData;
    //Request the book's ISBN, the sold copies, and the price per copy
    while (std::cin >> SalesData.ISBN >> SalesData.copies_sold >> SalesData.price) {
        const double revenue = SalesData.price * SalesData.copies_sold; //Calculate sales revenue
        //Print sales information along with revenue 
        std::cout << SalesData.ISBN << " " << SalesData.copies_sold << " " << revenue << std::endl;
    }
    return 0;
}
```

- _Write a program that reads two Sales_item objects that have the same ISBN and produces their sum._ **[(1.21)](https://github.com/Stachu72/CPP_Primer_Solutions/blob/main/Chapter%201/01.21.md)**:

```c++
#include <iostream>
#include <string>

struct Sales_data {
    std::string ISBN;
    unsigned copies_sold = 0;
    double price = 0.0;
};

int main()
{
    Sales_data SalesData1, SalesData2;
    if (std::cin >> SalesData1.ISBN >> SalesData1.copies_sold >> SalesData1.price) {       //Request the first sales data
        if (std::cin >> SalesData2.ISBN >> SalesData2.copies_sold >> SalesData2.price) {   //Request the second sales data
            if (SalesData1.ISBN == SalesData2.ISBN) { //Check if the books' ISBN match
                const unsigned totalSoldCopies = SalesData1.copies_sold + SalesData2.copies_sold;
                if (totalSoldCopies == 0) { //The total number of copies sold should be more than 0
                    std::cerr << "No sales!" << std::endl;
                    return -1;
                }
                const double totalRevenue = (SalesData1.price * SalesData1.copies_sold) + (SalesData2.price * SalesData2.copies_sold); //Calculate the total sales revenue
                const double pricePerCopy = (SalesData1.price + SalesData2.price) / 2; //Calculate the average price of the book
                std::cout << SalesData1.ISBN << " " << totalSoldCopies << " " << totalRevenue << " " << pricePerCopy << std::endl; //Print the sum of sales
                return 0;
            } 
            else {
                std::cerr << "The books' ISBN don't matching!" << std::endl;
                return -1;
            }
        } 
        else {
            std::cerr << "No input for the second sales data!" << std::endl;
            return -1;
        }
    } 
    else {
        std::cerr << "No input for the first sales data!" << std::endl;
        return -1;
    }
}
```

- _Write a program that reads several transactions for the same ISBN. Write the sum of all the transactions that were read._ **[(1.22)](https://github.com/Stachu72/CPP_Primer_Solutions/blob/main/Chapter%201/01.22.md)**:

```c++
#include <iostream>
#include <string>

struct Sales_data {
    std::string ISBN;
    unsigned copies_sold = 0;
    double price = 0.0;
};

int main()
{
    Sales_data CurrentSales, NextSales;
    if (std::cin >> CurrentSales.ISBN >> CurrentSales.copies_sold >> CurrentSales.price) { //Request for the first sales data
        unsigned totalSoldCopies = CurrentSales.copies_sold;
        double totalRevenue = CurrentSales.price;
        while (std::cin >> NextSales.ISBN >> NextSales.copies_sold >> NextSales.price) { //Keep requesting for the next sales data
            if (CurrentSales.ISBN == NextSales.ISBN) { //Check whether we're operating on the same book
                totalSoldCopies += NextSales.copies_sold;
                totalRevenue += NextSales.price * NextSales.copies_sold;
            }
            else {
                const double pricePerBook = totalRevenue / totalSoldCopies;
                std::cout << CurrentSales.ISBN << " " << totalSoldCopies << " " << totalRevenue << " " << pricePerBook << std::endl; //Print the sum of sales
                return 0;
            }
        }
        std::cerr << "No input?!" << std::endl;
        return -1;
    }
    else {
        std::cerr << "No input?!" << std::endl;
        return -1;
    }
}
```

- _Write a program that reads several transactions and counts how many transactions occur for each ISBN._ **[(1.23)](https://github.com/Stachu72/CPP_Primer_Solutions/blob/main/Chapter%201/01.23.md)**:

```c++
#include <iostream>
#include <string>

struct Sales_data {
    std::string ISBN;
    unsigned copies_sold = 0;
    double price = 0.0;
};

int main()
{
    Sales_data ReferencedSales, NextSales;
    if (std::cin >> ReferencedSales.ISBN) { //Request for the first ISBN to be referenced
        int count = 1;
        while (std::cin >> NextSales.ISBN) { //Request further ISBN numbers
            if (ReferencedSales.ISBN == NextSales.ISBN) { //Check whether the ISBNs match
                count++;
            }
            else {
                std::cout << "ISBN " << ReferencedSales.ISBN << " has occurred " << count << " times." << std::endl;
                count = 1;
                ReferencedSales = NextSales; //Change the ISBN to which it refers
            }
        }
    }
    else {
        std::cerr << "No input?!" << std::endl;
        return -1;
    }
}
```
