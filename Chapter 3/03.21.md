# Exercise 3.21
> Redo the first exercise from § 3.3.3 (p. 105) [(3.16)](https://github.com/Stachu72/CPP_Primer_Solutions/blob/main/Chapter%203/03.16.md) using iterators

## Solution

- **Code:**
    ```c++
    #include <iostream>
    #include <string>
    #include <vector>
    using std::cout; using std::endl;
    using std::string; using std::vector;
    /*
    * The program prints the size and contents of the specified vectors.
    * If the vector is empty or the element values are empty (strings only), the program prints a message about this.
    */
    int main() {
        //Prints information about v1
        vector<int> v1;
        cout << "Vector v1 | size: " << v1.size() << "  | element values: ";
        if (v1.empty())
            cout << "[no elements]" << " ";
        else
            for (auto it = v1.cbegin(); it != v1.cend(); ++it)
                cout << *it << " ";
        //Prints information about v2
        vector<int> v2(10);
        cout << endl << "Vector v2 | size: " << v2.size() << " | element values: ";
        if (v2.empty())
            cout << "[no elements]" << " ";
        else
            for (auto it = v2.cbegin(); it != v2.cend(); ++it)
                cout << *it << " ";
        //Prints information about v3
        vector<int> v3(10, 42);
        cout << endl << "Vector v3 | size: " << v3.size() << " | element values: ";
        if (v3.empty())
            cout << "[no elements]" << " ";
        else
            for (auto it = v3.cbegin(); it != v3.cend(); ++it)
                cout << *it << " ";
        //Prints information about v4
        vector<int> v4{10};
        cout << endl << "Vector v4 | size: " << v4.size() << "  | element values: ";
        if (v4.empty())
            cout << "[no elements]" << " ";
        else
            for (auto it = v4.cbegin(); it != v4.cend(); ++it)
                cout << *it << " ";
        //Prints information about v5
        vector<int> v5{10, 42};
        cout << endl << "Vector v5 | size: " << v5.size() << "  | element values: ";
        if (v5.empty())
            cout << "[no elements]" << " ";
        else
            for (auto it = v5.cbegin(); it != v5.cend(); ++it)
                cout << *it << " ";
        //Prints information about v6
        vector<string> v6{10};
        cout << endl << "Vector v6 | size: " << v6.size() << " | element values: ";
        if (v6.empty())
            cout << "[no elements]" << " ";
        else
            for (auto it = v6.cbegin(); it != v6.cend(); ++it)
                if (it->empty())
                    cout << "[empty]" << " ";
                else
                    cout << *it << " ";
        //Prints information about v7
        vector<string> v7{10, "hi"};
        cout << endl << "Vector v7 | size: " << v7.size() << " | element values: ";
        if (v7.empty())
            cout << "[no elements]" << " ";
        else
            for (auto it = v7.cbegin(); it != v7.cend(); ++it)
                if (it->empty())
                    cout << "[empty]" << " ";
                else
                    cout << *it << " ";
        return 0;
    }
    ```
