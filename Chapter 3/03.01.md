- _Write a program that uses a while to sum the numbers from 50 to 100._ **[(1.9)](https://github.com/Stachu72/CPP_Primer_Solutions/blob/main/Chapter%201/01.09.md)**:
```c++
#include <iostream>
using std::cout; using std::endl;

int main()
{
    int number = 50, sum = 0;
    while (number <= 100) {
        sum += number;
        number++;
    }
    cout << "The sum of the numbers from 50 to 100 inclusive, is: " << sum << endl;
    return 0;
}          
```

- _Write a program that reads two Sales_item objects that have the same ISBN and produces their sum._ **[(1.21)](https://github.com/Stachu72/CPP_Primer_Solutions/blob/main/Chapter%201/01.21.md)**:

```c++
#include <iostream>
#include <string>
using std::cin; using std::cout; using std::cerr; using std::endl; using std::string;

struct Sales_data {
    string ISBN;
    unsigned copies_sold = 0;
    double price = 0.0;
};

int main()
{
    Sales_data SalesData1, SalesData2;
    if (cin >> SalesData1.ISBN >> SalesData1.copies_sold >> SalesData1.price) {       //Request the first sales data
        if (cin >> SalesData2.ISBN >> SalesData2.copies_sold >> SalesData2.price) {   //Request the second sales data
            if (SalesData1.ISBN == SalesData2.ISBN) { //Check if the books' ISBN match
                const unsigned totalSoldCopies = SalesData1.copies_sold + SalesData2.copies_sold;
                if (totalSoldCopies == 0) { //The total number of copies sold should be more than 0
                    cerr << "No sales!" << endl;
                    return -1;
                }
                const double totalRevenue = (SalesData1.price * SalesData1.copies_sold) + (SalesData2.price * SalesData2.copies_sold); //Calculate the total sales revenue
                const double pricePerCopy = (SalesData1.price + SalesData2.price) / 2; //Calculate the average price of the book
                cout << SalesData1.ISBN << " " << totalSoldCopies << " " << totalRevenue << " " << pricePerCopy << endl; //Print the sum of sales
                return 0;
            } 
            else {
                cerr << "The books' ISBN don't matching!" << endl;
                return -1;
            }
        } 
        else {
            cerr << "No input for the second sales data!" << endl;
            return -1;
        }
    } 
    else {
        cerr << "No input for the first sales data!" << endl;
        return -1;
    }
}
```

- _Write a program that reads several transactions for the same ISBN. Write the sum of all the transactions that were read._ **[(1.22)](https://github.com/Stachu72/CPP_Primer_Solutions/blob/main/Chapter%201/01.22.md)**:

```c++
#include <iostream>
#include <string>
using std::cin; using std::cout; using std::cerr; using std::endl; using std::string;

struct Sales_data {
    string ISBN;
    unsigned copies_sold = 0;
    double price = 0.0;
};

int main()
{
    Sales_data CurrentSales, NextSales;
    if (cin >> CurrentSales.ISBN >> CurrentSales.copies_sold >> CurrentSales.price) { //Request for the first sales data
        unsigned totalSoldCopies = CurrentSales.copies_sold;
        double totalRevenue = CurrentSales.price;
        while (cin >> NextSales.ISBN >> NextSales.copies_sold >> NextSales.price) { //Keep requesting for the next sales data
            if (CurrentSales.ISBN == NextSales.ISBN) { //Check whether we're operating on the same book
                totalSoldCopies += NextSales.copies_sold;
                totalRevenue += NextSales.price * NextSales.copies_sold;
            }
            else {
                const double pricePerBook = totalRevenue / totalSoldCopies;
                cout << CurrentSales.ISBN << " " << totalSoldCopies << " " << totalRevenue << " " << pricePerBook << endl; //Print the sum of sales
                return 0;
            }
        }
        cerr << "No input?!" << endl;
        return -1;
    }
    else {
        cerr << "No input?!" << endl;
        return -1;
    }
}
```

- _Write a program that reads several transactions and counts how many transactions occur for each ISBN._ **[(1.23)](https://github.com/Stachu72/CPP_Primer_Solutions/blob/main/Chapter%201/01.23.md)**:

        while (cin >> NextSales.ISBN) { //Request further ISBN numbers
            if (ReferencedSales.ISBN == NextSales.ISBN) { //Check whether the ISBNs match
                count++;
            }
            else {
                cout << "ISBN " << ReferencedSales.ISBN << " has occurred " << count << " times." << endl;
                count = 1;
                ReferencedSales = NextSales; //Change the ISBN to which it refers
            }
        }
    }
    else {
        cerr << "No input?!" << endl;
        return -1;
    }
}
