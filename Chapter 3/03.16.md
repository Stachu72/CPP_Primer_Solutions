# Exercise 3.16
> Write a program to print the size and contents of the vectors from exercise [3.13](https://github.com/Stachu72/CPP_Primer_Solutions/blob/main/Chapter%203/03.13.md). Check whether your answers to that exercise were correct. If not, restudy § 3.3.1 (p. 97) until you understand why you were wrong.

## Solution

- **Code:**

```c++
#include <iostream>
#include <vector>
using std::cout; using std::endl; using std::string; using std::vector;

//Prints information about v1
    vector<int> v1;
    cout << "Vector v1 | size: " << v1.size() << " | element values: ";
    if (v1.empty())
        cout << "[no elements]" << " ";
    else
        for (const auto x : v1)
            cout << x << " ";
    //Prints information about v2
    vector<int> v2(10);
    cout << endl << "Vector v2 | size: " << v2.size() << " | element values: ";
    if (v2.empty())
        cout << "[no elements]" << " ";
    else
        for (const auto x : v2)
            cout << x << " ";
    //Prints information about v3
    vector<int> v3(10, 42);
    cout << endl << "Vector v3 | size: " << v2.size() << " | element values: ";
    if (v3.empty())
        cout << "[no elements]" << " ";
    else
        for (const auto x : v3)
            cout << x << " ";
    //Prints information about v4
    vector<int> v4{10};
    cout << endl << "Vector v4 | size: " << v2.size() << " | element values: ";
    if (v4.empty())
        cout << "[no elements]" << " ";
    else
        for (const auto x : v4)
            cout << x << " ";
    //Prints information about v5
    vector<int> v5{10, 42};
    cout << endl << "Vector v5 | size: " << v2.size() << " | element values: ";
    if (v5.empty())
        cout << "[no elements]" << " ";
    else
        for (const auto x : v5)
            cout << x << " ";
    //Prints information about v6
    vector<string> v6{10};
    cout << endl << "Vector v6 | size: " << v2.size() << " | element values: ";
    if (v6.empty())
        cout << "[no elements]" << " ";
    else
        for (const auto x : v6)
            if (x.empty())
                cout << "[empty]" << " ";
            else
                cout << x << " ";
    //Prints information about v7
    vector<string> v7{10, "hi"};
    cout << endl << "Vector v7 | size: " << v2.size() << " | element values: ";
    if (v7.empty())
        cout << "[no elements]" << " ";
    else
        for (const auto x : v7)
            if (x.empty())
                cout << "[empty]" << " ";
            else
                cout << x << " ";
    return 0;
```
